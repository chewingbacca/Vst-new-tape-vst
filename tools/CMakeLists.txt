# Offline core tools (no JUCE): quick iteration on voices and effects.
foreach(t voice_render fx_render core_smoke)
  add_executable(${t} core/${t}.cpp)
  target_link_libraries(${t} PRIVATE automata_core)
  target_include_directories(${t} PRIVATE core)
endforeach()

# Offline host harness around the real AudioProcessor (links the plugin's shared code).
if(AUTOMATA_BUILD_PLUGIN)
  function(automata_host_tool name)
    add_executable(${name} ${ARGN})
    target_include_directories(${name} PRIVATE host $<TARGET_PROPERTY:Automata,INCLUDE_DIRECTORIES>)
    target_compile_definitions(${name} PRIVATE $<TARGET_PROPERTY:Automata,COMPILE_DEFINITIONS>)
    target_link_libraries(${name} PRIVATE Automata)
  endfunction()
  automata_host_tool(automata_render host/automata_render.cpp host/HostHarness.cpp)
  automata_host_tool(automata_snapshot host/automata_snapshot.cpp host/HostHarness.cpp)
  automata_host_tool(automata_render_tests host/automata_render_tests.cpp host/HostHarness.cpp)
  add_test(NAME render_tests COMMAND automata_render_tests --quick)
endif()
